model AsaasWebhookLog {

const ASAAS_API_URL = process.env.ASAAS_API_URL ?? 'https://sandbox.asaas.com/api/v3'
const ASAAS_API_KEY = process.env.ASAAS_API_KEY ?? ''

async function asaasFetch(path: string, init: RequestInit = {}) {
  const res = await fetch(`${ASAAS_API_URL}${path}`, {
    ...init,
    headers: {
      'Content-Type': 'application/json',
      access_token: ASAAS_API_KEY,
      ...init.headers,
    },
  })
  if (!res.ok) {
    const err = await res.json().catch(() => ({}))
    throw new Error(JSON.stringify(err))
  }
  return res.json()
}

📄 app/api/asaas/webhook/route.ts — fluxo esperado
export const dynamic = 'force-dynamic'

export async function POST(req: NextRequest) {
  // 1. Ler body e validar estrutura
  const body = await req.json()
  if (!body?.event || !body?.payment) return Response.json({ received: true })

  // 2. Validar header asaas-access-token
  const token = req.headers.get('asaas-access-token')
  if (token !== process.env.ASAAS_WEBHOOK_TOKEN) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const { event, payment } = body

  // 3. Gravar log (idempotência)
  const log = await prisma.asaasWebhookLog.create({
    data: { asaasEvent: event, paymentId: payment.id, payload: body }
  })

  // 4. Verificar se já foi processado
  const dup = await prisma.asaasWebhookLog.findFirst({
    where: { paymentId: payment.id, asaasEvent: event, processed: true }
  })
  if (dup) return Response.json({ received: true }) // skip

  // 5. Encontrar User (pelo externalReference ou asaasCustomerId)
  const user = await prisma.user.findFirst({
    where: { OR: [{ id: payment.externalReference }, { asaasCustomerId: payment.customer }] }
  })
  if (!user) return Response.json({ received: true })

  // 6. Processar evento → atualizar isActive / planType
  // ...

  // 7. Marcar log como processado
  await prisma.asaasWebhookLog.update({ where: { id: log.id }, data: { processed: true } })

  return Response.json({ received: true })
}

⚡ Considerações de Coexistência — Stripe & Asaas

🔄 Estratégia de Coexistência

  • ✓ Stripe permanece ativo para pagamentos internacionais (cartão internacional)
  • novo Asaas cobre mercado brasileiro (PIX, boleto, cartão nacional)
  • ~ O modelo User terá campos de ambos: stripeCustomerId + asaasCustomerId
  • ~ O planType é atualizado por qualquer gateway (webhook Stripe ou Asaas)
  • ⚠ Cuidado para não ter assinaturas ativas em ambos para o mesmo usuário simultaneamente
  • ⚠ Definir claramente na UI qual gateway é usado (ex: "Pagar com PIX" → Asaas, "Pagar com cartão internacional" → Stripe)
